create database applicationdb;

use applicationdb;

create table users
(
    user_id int auto_increment primary key,
    name varchar(40) not null,
    email varchar(120) not null,
    password varchar(20) not null,
    phone varchar(10) not null,
    pic varchar(100),
    user_type varchar(1) not null default "S"
);

create table students
(
    student_id int auto_increment primary key,
    name varchar(60) not null,
    fath_name varchar(60) not null,
    moth_name varchar(60) not null,
    age int not null,
    email varchar(100) not null,
    phone varchar(10) not null,
    user_id int not null,
    constraint fk_students_users foreign key (user_id) references users (user_id)
);

create table studentPic
(
    student_pic_id int auto_increment primary key,
    student_id int not null,
    pic_path varchar(150) not null,
    main_pic bool default false,
    constraint fk_studentPic_students foreign key (student_id) references students (student_id)
);
